Action and predicate schemas ​

Extracted from the current checkout. Source links navigate current main; SHA-256 identifies the extracted bytes, which may differ from main. Source comments describe the declarations; see the contract map for runtime boundaries.

The schema below includes private component schemas so every nested target and predicate shape is visible. The validator compiles the same action/predicate objects used by the tools.

ts
import { Compile } from "typebox/compile";
import { Type } from "@earendil-works/pi-ai";
import { fail } from "./helper.ts";

// One closed schema for model-facing validation and the executor boundary. Never strip fields.
const object: typeof Type.Object = (properties, options) => Type.Object(properties, { ...options, additionalProperties: false });

export const MatchSchema = Type.Union([Type.String(), object({ contains: Type.String() }), object({ regex: Type.String() })], {
	description: 'Exact text (case-insensitive), {"contains": "..."}, or {"regex": "..."}',
});
const TargetSchema = Type.Union(
	[
		object({ id: Type.String({ description: 'Element id from the latest observation, e.g. "e14"' }) }),
		object({
			role: Type.Optional(Type.String()),
			name: Type.Optional(MatchSchema),
			within: Type.Optional(Type.String({ description: "id of an ancestor" })),
			index: Type.Optional(Type.Number({ description: "0-based, when several match" })),
		}),
		object({ point: object({ x: Type.Number(), y: Type.Number() }, { description: "0-1 from the top-left; only from a screenshot" }) }),
	],
	{ description: "An element: {id}, a locator {role, name}, or {point}" },
);
const TextTargetSchema = object({ app: Type.Optional(Type.String()), window: Type.Optional(MatchSchema), element: Type.Optional(TargetSchema) });
const leaves = [
	object({ app_frontmost: Type.String() }),
	object({ window: object({ app: Type.Optional(Type.String()), title: MatchSchema }) }),
	object({ exists: TargetSchema }),
	object({ absent: TargetSchema }),
	object({ focused: TargetSchema }),
	object({ value: object({ target: TargetSchema, is: MatchSchema }) }),
	object({ text: object({ target: Type.Optional(TextTargetSchema), contains: Type.Optional(Type.String()), equals: Type.Optional(Type.String()) }) }),
	object({ changed: Type.Union([Type.Literal("any"), Type.Literal("window"), Type.Literal("focus"), TargetSchema]) }),
	object({ visual: Type.String({ description: "Checked by you from a screenshot" }) }),
];
const LeafSchema = Type.Union(leaves);
export const PredicateSchema = Type.Union([...leaves, object({ all: Type.Array(LeafSchema) }), object({ any: Type.Array(LeafSchema) }), object({ not: LeafSchema })], {
	description: "What is observably true afterwards",
});
const ViaSchema = Type.Union([Type.Literal("auto"), Type.Literal("ax"), Type.Literal("gui")]);
export const ActionSchema = Type.Union([
	object({ type: Type.Literal("open_app"), app: Type.String({ description: "Name or bundle id" }) }),
	object({ type: Type.Literal("focus_window"), app: Type.String({ minLength: 1 }), window: MatchSchema }),
	object({ type: Type.Literal("menu"), path: Type.Array(Type.String(), { description: '["File", "New"]' }), app: Type.Optional(Type.String()) }),
	object({ type: Type.Literal("press"), target: TargetSchema, via: Type.Optional(ViaSchema) }, { description: "AX activation of a target; keyboard shortcuts use type=key, key, modifiers." }),
	object({ type: Type.Literal("click"), target: TargetSchema, button: Type.Optional(Type.Union([Type.Literal("left"), Type.Literal("right")])), count: Type.Optional(Type.Number()) }),
	object({ type: Type.Literal("set_value"), target: TargetSchema, value: Type.String(), via: Type.Optional(ViaSchema) }),
	object({ type: Type.Literal("type"), text: Type.String(), target: Type.Optional(TargetSchema), via: Type.Optional(ViaSchema) }, { description: "Text entry. Default: inserted through accessibility at the caret in small timed writes paced like fast typing (CARET_TYPE_WPM, 200 wpm; 0 is one write) when the field allows it, else keystrokes. via gui: real keystrokes, one key per character (one-box-per-character codes). via ax: accessibility insertion only; fails with NOT_ACTIONABLE when the field refuses it, no keystroke fallback." }),
	object({ type: Type.Literal("key"), key: Type.String({ description: 'e.g. "return", "n", "tab"' }), modifiers: Type.Optional(Type.Array(Type.String({ description: "cmd, shift, opt, ctrl, fn" }))) }),
	object({ type: Type.Literal("scroll"), target: TargetSchema, dy: Type.Number({ description: "negative scrolls down" }), dx: Type.Optional(Type.Number()) }),
	object({ type: Type.Literal("wait"), until: PredicateSchema, timeout_ms: Type.Optional(Type.Number({ maximum: 5000 })) }),
	object({ type: Type.Literal("verify"), check: PredicateSchema }),
	object(
		{ type: Type.Literal("paste_secret"), target: Type.Optional(TargetSchema) },
		{ description: "Secret handoff tasks only: paste the copied secret into the target field (or the focused one) without seeing it." },
	),
]);


const executionValidator = Compile(object({ actions: Type.Array(ActionSchema), expect: PredicateSchema }));

/** Validate the entire batch before snapshots, recorder reads, journals or mutations. */
export function validateExecution(actions: unknown, expect: unknown): void {
 if (!executionValidator.Check({ actions, expect })) {
  // Do not echo input text, which can include secrets or user drafts.
  throw fail("INVALID_ACTION", 'Invalid action or predicate fields. press activates an AX target (type, target, optional via); to press Escape use {"type":"key","key":"escape"}; shortcuts use key with modifiers. Never add key/modifiers to press or a target to key. Use only fields declared for that variant, including nested targets and predicates.');
 }
}
