RUN_MAC_TASK_V2 ​

src/supervisor.ts · SHA-256: d6e662c636437e0623a584ed2af9aae4c5d0572b32ec1b7e39f2ab2fceea064f

ts
const RUN_MAC_TASK_V2 = {
	description:
		"Delegate a task on the user's Mac to the Pi computer-use agent (it reads the accessibility tree, clicks, types, uses menus). " +
		"Give a self-contained goal and success_criteria as observable end states. Returns JSON: status (success, partial, failed, needs_user, " +
		"blocked_safety, budget_exhausted, interrupted), summary, question, evidence, task_id. On needs_user, ask the user the question, then call " +
		"run_mac_task again with resume {trace_id, answer} (and any goal). For an explicit send request, constraints.allow may include send. Hard app blocks, live work and uncertain destinations cannot be overridden with constraints.allow. " +
		"For a multi-step job, pass plan as a short checklist.",
	parameters: {
		type: "object",
		properties: {
			goal: { type: "string", description: "Self-contained, imperative: which app, what to do" },
			success_criteria: { type: "array", items: { type: "string" }, description: "Observable end states, plain language" },
			success_checks: { type: "array", items: { type: "object" }, description: `Optional machine checks. ${PREDICATE_HELP}` },
			plan: { type: "array", items: { type: "string" }, description: "Optional checklist of steps, in order" },
			constraints: {
				type: "object",
				properties: {
					allow: { type: "array", items: { type: "string", enum: ["send", "purchase", "delete", "submit", "publish", "credential", "irreversible"] } },
					forbid: { type: "array", items: { type: "string" } },
				},
			},
			app_hints: {
				type: "object",
				properties: { apps: { type: "array", items: { type: "string" } }, window: { type: "string" }, files: { type: "array", items: { type: "string" } }, notes: { type: "string" } },
			},
			budget: { type: "object", properties: { max_steps: { type: "number" }, max_seconds: { type: "number" } } },
			resume: { type: "object", properties: { trace_id: { type: "string" }, answer: { type: "string" } }, required: ["trace_id", "answer"] },
			secret: {
				type: "object",
				description:
					"Set for any task that handles an API key, token or password value; nobody (Pi, its model, logs, recordings) sees the value. " +
					'mode "user" (default choice): Pi does everything except the clicks that reveal, copy or paste the secret, and hands those to the user. ' +
					'mode "handoff": only when the user asks caret to move the secret itself; Pi presses the source\'s Copy button and pastes with paste_secret.',
				properties: { mode: { type: "string", enum: ["user", "handoff"] }, name: { type: "string", description: "What the secret is called, e.g. HF_TOKEN" } },
				required: ["mode"],
			},
			effort: {
				type: "string",
				enum: ["medium", "high", "xhigh"],
				description: "Optional reasoning effort for Pi. Leave unset (caret picks: medium for one or two plain clicks, xhigh for precision or long jobs); set it to override.",
			},
		},
		required: ["goal"],
	},
};
