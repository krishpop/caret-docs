DeltaEntry ​

ts
export type DeltaEntry = {
	op: DeltaOp;
	key?: string;
	ax_path?: string;
	role?: string;
	/** value_changed: which attribute (absent = "value"). */
	attr?: "value" | "title" | "label";
	from?: string | null;
	to?: string | null;
	/** sha256 of the value, used instead of raw text when it is long or was truncated. */
	from_hash?: string;
	to_hash?: string;
	to_len?: number;
	/** window_opened/closed */
	app?: string;
	title?: string;
	window_id?: number;
};
