TaskSpec ​

ts
export type TaskSpec = {
	trace_id: string;
	goal: string;
	success_criteria?: string[];
	success_checks?: Predicate[];
	constraints?: { allow?: RiskClass[]; forbid?: string[]; no_secure_fields?: boolean };
	app_hints?: { apps?: string[]; window?: string; files?: string[]; notes?: string };
	budget?: Partial<Budget>;
	/** Checklist to work through; plain strings become pending items p1..pN. */
	plan?: (string | { id?: string; text: string })[];
	resume?: { trace_id: string; answer: string };
	/**
	 * The task moves a secret (API key, token, password) whose value nobody may see (docs/secret-handoff.md).
	 * "user": the agent does every step except revealing, copying and pasting the value, which it hands to the user.
	 * "handoff": the agent clicks the source's Copy button and pastes with paste_secret, never reading the value.
	 * Either way: no screenshots, no step records, secret-shaped text hidden from the model.
	 */
	secret?: { mode: "user" | "handoff"; name?: string };
	/** Reasoning effort override (pi-ai thinking level, e.g. "medium", "high", "xhigh"); clamped to the model. Default: src/effort.ts rules. */
	effort?: "off" | "minimal" | "low" | "medium" | "high" | "xhigh" | "max";
};

TaskProgress ​

ts
/**
 * What a progress UI (command bar progress bar + checklist) or a `task_status` tool reads while a
 * task runs. Derived from the caret.task document only, so it is cheap and survives restarts.
 */
export type TaskProgress = {
	trace_id: string;
	task_id: string;
	phase: TaskPhase;
	status?: TaskStatus;
	goal: string;
	/** Current batch intent ("create a new document"), or the active plan item. */
	current?: string;
	/** 0-1: plan items done / total when there is a plan, else steps / max_steps. */
	fraction: number;
	plan: PlanItem[];
	steps: StepLog[];
	usage: Omit<Usage, "tokens_in" | "tokens_out">;
	budget: Budget;
	/** Foreground (CGEvent clicks/typing that take focus) vs background (AX, menus) for the last batch. */
	mode?: "foreground" | "background";
	summary?: string;
	question?: string;
	updated_at: number;
	/** Set while the model call is failing and Pi is backing off before retrying (e.g. provider connection errors). */
	model_retry?: { attempt: number; error: string; next_in_s: number };
};
