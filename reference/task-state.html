/**
 * The `caret.task` document (spec 7.4): one per task conversation, the source of truth for the task's
 * status, step log, counters and plan. TaskResult and TaskProgress are both derived from it.
 */
export type TaskState = {
	/** Cleared only after the runner confirms helper cleanup; recovered on cancelAll after restart. */
	secret_cleanup_pending?: boolean;
	spec: TaskSpec;
	phase: TaskPhase;
	status?: TaskStatus;
	summary?: string;
	question?: string;
	error?: CaretError;
	started_at: number;
	updated_at: number;
	finished_at?: number;
	steps: StepLog[];
	counters: { steps: number; actions: number; images: number };
	approaches: { label: string; batches: number; failed: boolean; streak: number }[];
	last_batch?: { approach: string; failed: boolean; expect: "pass" | "fail" | "unverified" };
	no_change_streak: number;
	/** Fingerprints of the actions of the last two failed batches (loop detection). */
	failed_fps: string[][];
	recorded_batches: string[];
	actions_taken: ActionRecord[];
	evidence: Evidence[];
	artifacts: Artifact[];
	approvals: ApprovalRecord[];
	plan: PlanItem[];
	last_obs_id: number;
	last_obs_excerpt?: string;
	finish_rejections: number;
	/** needs_user answers submitted so far. */
	resumes: number;
	mode?: "foreground" | "background";
	/** Trajectory recording ids (docs/SCHEMA_LITE.md), set at the first act; `written` once the task record is out. */
	rec?: { episode_id: string; task_id: string; started_at: number; first_step?: string; last_step?: string; steps: number; written?: boolean };
};

TaskRunner ​

ts
/**
 * Per-task Pi conversations (spec 7.3) with an async contract: `start()` returns a task id at once,
 * `progress()` reads the caret.task document (what a command-bar progress bar / checklist polls),
 * `wait()` resolves with the TaskResult. `run()` = start + wait, for the synchronous run_mac_task.
 */
export type TaskRunner = {
	/**
	 * `thinkingLevel` (from src/effort.ts, chosen by the caller) sets the task's reasoning effort; default: chooseEffort(spec).
	 * `model` runs this task on another model than the runner's default (a resumed task keeps the model it runs on: the one it started on, or the fallback after a failover).
	 */
	start(spec: TaskSpec, opts?: { thinkingLevel?: EffortLevel; model?: ModelRef }): Promise<{ task_id: string; trace_id: string; resumed: boolean }>;
	progress(idOrTrace: string): Promise<TaskProgress | undefined>;
	wait(idOrTrace: string): Promise<TaskResult>;
	run(spec: TaskSpec, opts?: { thinkingLevel?: EffortLevel; model?: ModelRef }): Promise<TaskResult>;
	cancel(idOrTrace: string, reason?: string): Promise<boolean>;
	/** Switches an unfinished task to another model; its next attempt (even a retry in backoff) uses it. */
	failover(idOrTrace: string, model: ModelRef): Promise<boolean>;
	/** Aborts every unfinished task and marks it interrupted; returns their trace ids. */
	cancelAll(reason: string): Promise<string[]>;
};
